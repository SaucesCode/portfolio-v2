## 2. Information Architecture & Canonical URL Strategy

### 2.1 Configurable Production Domain
To avoid hardcoding an unverified or unowned domain, the canonical base URL is configured dynamically via environment variables with a safe fallback:

```typescript
// lib/seo.ts
export const getBaseUrl = (): string => {
  if (process.env.NEXT_PUBLIC_SITE_URL) {
    return process.env.NEXT_PUBLIC_SITE_URL.replace(/\/$/, '');
  }
  if (process.env.VERCEL_URL) {
    return `https://${process.env.VERCEL_URL}`;
  }
  return 'http://localhost:3000';
};
```

### 3.2 Dynamic Case Study Metadata (`app/work/[slug]/page.tsx`)
```typescript
import type { Metadata } from 'next';
import { getProjectBySlug } from '@/lib/projects';
import { getBaseUrl } from '@/lib/seo';

export async function generateMetadata({
  params,
}: {
  params: Promise<{ slug: string }>;
}): Promise<Metadata> {
  const { slug } = await params;
  const project = await getProjectBySlug(slug);
  const baseUrl = getBaseUrl();

  if (!project) {
    return { title: 'Project Not Found' };
  }

  const canonicalUrl = `${baseUrl}/work/${slug}`;

  return {
    title: `${project.title} — Case Study`,
    description: project.summary,
    alternates: {
      canonical: canonicalUrl,
    },
    openGraph: {
      title: `${project.title} | Case Study`,
      description: project.summary,
      url: canonicalUrl,
      type: 'article',
      images: [
        {
          url: `${baseUrl}/work/${slug}/og-image.png`,
          width: 1200,
          height: 630,
          alt: project.title,
        },
      ],
    },
  };
}
```

## 4. XML Sitemap & Robots Architecture

### 4.1 Crawler Directives (`app/robots.ts`)
> [!IMPORTANT]
> **CRITICAL CRAWLER FIX:** Never block `/_next/*` in `robots.txt`! Modern search engine crawlers (Googlebot) render pages identically to a real browser. Blocking `/_next/*` prevents crawlers from fetching JavaScript hydration bundles, stylesheets, and fonts, resulting in unrendered pages, mobile-unfriendliness flags, and severe indexing penalties.

```typescript
import type { MetadataRoute } from 'next';
import { getBaseUrl } from '@/lib/seo';

export default function robots(): MetadataRoute.Robots {
  const baseUrl = getBaseUrl();

  return {
    rules: {
      userAgent: '*',
      allow: '/',
      disallow: ['/api/*'], // Disallow private internal API routes only
    },
    sitemap: `${baseUrl}/sitemap.xml`,
  };
}
```

### 4.2 Dynamic XML Sitemap (`app/sitemap.ts`)
Fabricated `lastModified` timestamps (such as generating `new Date().toISOString()` on every request) harm search credibility. In our architecture:
- Static routes specify actual meaningful content revision dates or omit `lastModified` when unknown.
- Project routes read the actual repository or content update date.
- All genuinely publishable service routes are explicitly indexed.

```typescript
import type { MetadataRoute } from 'next';
import { getBaseUrl } from '@/lib/seo';

export default async function sitemap(): Promise<MetadataRoute.Sitemap> {
  const baseUrl = getBaseUrl();

  // Core Static Routes (Meaningful Content Revision Dates)
  const staticRoutes: MetadataRoute.Sitemap = [
    { url: `${baseUrl}`, changeFrequency: 'monthly', priority: 1.0 },
    { url: `${baseUrl}/work`, changeFrequency: 'weekly', priority: 0.9 },
    { url: `${baseUrl}/services`, changeFrequency: 'monthly', priority: 0.8 },
    { url: `${baseUrl}/about`, changeFrequency: 'monthly', priority: 0.7 },
    { url: `${baseUrl}/contact`, changeFrequency: 'monthly', priority: 0.8 },
  ];

  // Genuinely Publishable Dedicated Service Pages
  const serviceRoutes: MetadataRoute.Sitemap = [
    {
      url: `${baseUrl}/services/wordpress-fse`,
      changeFrequency: 'monthly',
      priority: 0.8,
    },
    {
      url: `${baseUrl}/services/full-stack-systems`,
      changeFrequency: 'monthly',
      priority: 0.8,
    },
    {
      url: `${baseUrl}/services/e-commerce`,
      changeFrequency: 'monthly',
      priority: 0.8,
    },
  ];

  // Verified Project Slugs (Populated from static project registry)
  const verifiedProjects = [
    { slug: 'thread-and-form', lastModified: '2026-10-15' },
    { slug: 'portfolio-cms', lastModified: '2026-09-20' },
    { slug: 'kinetix-athletics', lastModified: '2026-08-10' },
    { slug: 'horology-landing', lastModified: '2026-07-05' },
    { slug: 'valence-capital', lastModified: '2026-06-18' },
  ];

  const projectRoutes: MetadataRoute.Sitemap = verifiedProjects.map((p) => ({
    url: `${baseUrl}/work/${p.slug}`,
    lastModified: p.lastModified,
    changeFrequency: 'monthly',
    priority: 0.8,
  }));

  return [...staticRoutes, ...serviceRoutes, ...projectRoutes];
}
```

## 6. Google Search Console Protocol & Indexing Audit

Following deployment to production:
1. **Property Verification:** Add domain property in Google Search Console using DNS TXT record or HTML meta tag.
2. **Sitemap Submission:** Submit `/sitemap.xml` directly within Search Console. Verify status returns `Success`.
3. **URL Inspection:** Inspect root URL `/` and top case studies using the URL Inspection tool to confirm:
   - Google can fetch and render page assets (confirming `/_next/` resources load cleanly).
   - Structured data (JSON-LD) parses with zero errors or warnings.
   - User-declared canonical matches Google-selected canonical.
4. **Core Web Vitals Monitoring:** Monitor field data under the "Core Web Vitals" report.
